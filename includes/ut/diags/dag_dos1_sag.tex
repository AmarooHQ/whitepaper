\input{dag-styles}%
\providecommand\DagNodeDist{0.8cm}
\newcommand\TikzDagSimple{
    \providecommand\AbbrHChain{H}
    \providecommand\AbbrAChain{A}
    \node (prevDots) [yshift=0cm, align=center] {$\cdots$};

    \node (block0) [block, align=center, right = of prevDots] {$\AbbrHChain_{i}$ \\ \footnotesize{$\Sigma_{w} = 0$}};
    \draw [arrowParent] (block0) -- (prevDots);
    % \node [anchor=center, above = of block0.south] {$\Sigma_{w} = 0$};

    % \node (spacing) [block, above = of block0, yshift=100pt] {spacing debug node};

    \node (forkPoint) [right = of block0, xshift=20pt] {};

    % start of attacker's chain
    \xdef\prevNodeBeforeLoop{block0}
    \xdef\prevNode{\prevNodeBeforeLoop}
    \xdef\nodePosDir{above}
    \xdef\nodePosRel{forkPoint}
    \foreach \a in {1,...,7} {
        \node (blockA\a) [block, align=center, \nodePosDir = of \nodePosRel] {$\AbbrAChain_{i+\a}$ \\ \footnotesize{$\Sigma_{w} = \a$}};
        \draw [arrowParent] (blockA\a) -- (\prevNode);
        \xdef\prevNode{blockA\a}
        \xdef\nodePosDir{right}
        \xdef\nodePosRel{\prevNode}
    }

    % honest chain
    \xdef\prevNode{\prevNodeBeforeLoop}
    \xdef\nodePosDir{below}
    \xdef\nodePosRel{forkPoint}
    \xdef\lastHBlock{2}

    \node (blockH1) [block, align=center, \nodePosDir = of \nodePosRel] {$\AbbrHChain_{i+1}$ \\ \footnotesize{$\Sigma_{w} = 1$}};
    \draw [arrowParent] (blockH1) -- (\prevNode);

    \foreach \h in {2,...,4} {
        \pgfmathtruncatemacro\prevH{int(\h-1)}
        \pgfmathtruncatemacro\prevA{int((\h-1)*2)}
        \pgfmathtruncatemacro\sigmawH{\prevH + \prevA}
        \xdef\prevNodes{blockH\prevH,blockA\prevA}
        \xdef\nodePosDir{right}
        \xdef\nodePosRel{blockH\prevH -| blockA\prevA.east}
        \xdef\lastHBlock{\h}

        \node (blockH\h) [block, align=center, \nodePosDir = of \nodePosRel] {$\AbbrHChain_{i+\h}$ \\ \footnotesize{$\Sigma_{w} = \sigmawH$}};

        \foreach \p in \prevNodes {
            \draw [arrowParent] (blockH\h) -- (\p);
        }
    }

    \node (labelA) [align=center] at (blockA1 -| block0) {Attacker's \\ Blocks};
    \node (labelH) [align=center] at (blockH1 -| block0) {Honest \\ Blocks};

    \node (labelALineHeight) [below = of labelA, yshift=0.85*\DagNodeDist] {};
    \node (labelHLineHeight) [above = of labelH, yshift=-0.85*\DagNodeDist] {};


    \xdef\xs{0cm}
    \xdef\topNode{blockA1}
    \xdef\botNode{blockH1}
    \xdef\leftMostNode{prevDots}
    \xdef\rightMostNode{blockH\lastHBlock}

    \draw [dashed] (labelALineHeight -| \leftMostNode.west) -- (labelALineHeight -| \rightMostNode.east);
    \draw [dashed] (labelHLineHeight -| \leftMostNode.west) -- (labelHLineHeight -| \rightMostNode.east);

    \node (topLeft) [above = of \topNode -| \leftMostNode.west, minimum width=0, xshift=\xs] {};
    \node (botLeft) [below = of \botNode -| \leftMostNode.west, minimum width=0, xshift=\xs] {};
    \node (topRight) [above = of \topNode -| \rightMostNode.east, minimum width=0, xshift=\xs] {};
    \node (botRight) [below = of \botNode -| \rightMostNode.east, minimum width=0, xshift=\xs] {};
    % \draw [arrowTime] (topLeft) node () [anchor=west, yshift=-10pt] {\textit{Time}} -- (topRight);
    \draw [arrowTime] (botLeft) node () [anchor=west, yshift=-10pt] {\textit{Time}} -- (botRight);
}%
\begin{tikzpicture}[node distance=\DagNodeDist]
    \TikzDagSimple
\end{tikzpicture}
\end{document}
