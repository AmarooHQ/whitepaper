\newcommand{\bissectDraw}[6][]{%
    \path[#1] let
        \p1 = ($(#3)!1cm!(#2)$),
        \p2 = ($(#3)!1cm!(#4)$),
        \p3 = ($(\p1) + (\p2) - (#3)$)
    in
        ($(#3)!#6!(\p3)$) -- ($(\p3)!#5!(#3)$) ;
    }
% p4: the corner
% p1,2: same length towards the other corners
    % \p5 = ($((\p3) + (((\p1) + (\p2))!1pt!(\p3)))$)
    % \p5 = ($(\p3)$)
% ($(#3)!#6!(\p3)$) -- ($(\p3)!#5!(#3)$) ;
        % \p5 = ($((\p1) + (\p2))$),
\newcommand{\reflectLine}[5][]{%
    \ifnum 1=\DrawPoRMirrors
    \path[#1] let
        \p4 = ($(#3)$),
        \p1 = ($(#3)!1cm!(#2)$),
        \p2 = ($(#3)!1cm!(#4)$),
        \p3 = ($(\p1) + (\p2) - (\p4)$),
        \p5 = ($(\p4)!.5pt!180:(\p3)$)
    in
    ($(\p5)!#5!90:(\p3)$)
        % -- node [rectangle, draw] {} (\p3)
        -- ($(\p5)!-#5!90:(\p3)$);
    \fi
    }
