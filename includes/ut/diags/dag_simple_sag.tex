\newcommand\TikzDagSimple{
    \providecommand\AbbrChain{B}
    \node (prevDots) [yshift=0cm] {$\vdots$};

    \node (block1) [block, below = of prevDots] {$\AbbrChain_{i}$};
    \draw [arrowParent] (block1) -- (prevDots);

    \node (block2mid) [below = of block1] {};
    \node (block2/1) [block, left = of block2mid] {$\AbbrChain_{i+1,1}$};
    \node (block2/2) [block, right = of block2mid] {$\AbbrChain_{i+1,2}$};
    \draw [arrowParent] (block2/1) -- (block1);
    \draw [arrowParent] (block2/2) -- (block1);

    \node (block3) [block, below = of block2mid] {$\AbbrChain_{i+2}$};
    \draw [arrowParent] (block3) -- (block2/1);
    \draw [arrowParent] (block3) -- (block2/2);

    \xdef\xs{0cm}

    \node (topLeft) [left = of prevDots -| block2/1, minimum width=0, xshift=\xs] {};
    \node (botLeft) [left = of block3.south -| block2/1, minimum width=0, xshift=\xs] {};
    \draw [arrowTime] (topLeft) node () [anchor=west, yshift=-10pt] {\textit{Time}} -- (botLeft);
}%
