\begin{tikzpicture}[node distance=0.7cm]

    \node (btcChain) [chainName, rotate=90] {Bitcoin};
    \node (ethChain) [chainName, below of=btcChain, yshift=-3cm, rotate=90] {Ethereum};

    \foreach \net/\oNet in {btc/eth,eth/btc} {
        \node (\net Dots) [right of=\net Chain, xshift=0.3cm] {$\cdots$};
    }

    \foreach \x in {1,...,3} {
        \pgfmathtruncatemacro\r{round(\x-1)}
        \pgfmathtruncatemacro\ethLastEnd{round((\x-1)*40)}
        \pgfmathtruncatemacro\ethBStart{round((\x-1)*40+1)}
        \pgfmathtruncatemacro\ethBEnd{round((\x-1)*40+40)}

        %%%%%%%%%
        % bitcoin
        %%%%%%%%%

        \ifnum 1=\x
        \xdef\ys{0}
        \xdef\prevNode{btcDots}
        \xdef\oPrev{ethDots}
        \else
        % \xdef\ys{-1cm}
        \xdef\prevNode{btc\r}
        \xdef\oPrev{ethE\r}
        \fi

        \node (btc\x) [block, right = of \prevNode -| \oPrev, yshift=\ys] {$\MakeUppercase{btc}_{i+\x}$};
        \draw [arrowParent] (btc\x) -- (\prevNode);

        % btc state
        \draw [braceRL] (btc\x.west |- btc\x.north) --
            node (btcState\x) [anchor=south, xshift=30pt, yshift=6pt, text width=110pt] {\footnotesize{
                \textsc{STATE} \\
                Headers: $ETH_{0 \cdots j+\ethLastEnd}$ \\
                Latest PoR: $BTC_{i+\r}$ \\
                %% $\Sigma$ Weight: $\Sigma^{j+\ethLastEnd}_{k=0} \text{W}(ETH_{k})$
                }}
            (btc\x.east |- btc\x.north);

        % reflection arrow, only on nodes 2 and beyond
        \ifnum 1<\x
        \draw [arrowRefl] (btc\x) --
            node [anchor=center, above, sloped] {\scriptsize{include 40 headers}}
            node [anchor=center, below, sloped] {\scriptsize{and 40 PoRs}}
            (ethAll\r);
        \fi

        %%%%%%%%%%
        % ethereum
        %%%%%%%%%%

        \xdef\ys{0}

        \ifnum 1=\x
        \xdef\prevNode{ethDots}
        \xdef\oPrev{btc1}
        \else
        \xdef\prevNode{ethE\r}
        \xdef\oPrev{btc\x}
        \fi

        \ifnum 3>\x

        \node (ethS\x) [block, right = of \prevNode -| \oPrev, xshift=-0.5cm] {$\MakeUppercase{eth}_{j+\ethBStart}$};
        \draw [arrowParent] (ethS\x) -- (\prevNode);

        \node (ethDots\x) [dots, right = of ethS\x, xshift=-0.4cm, sloped] {$\cdots$};
        \draw [arrowParent] (ethDots\x) -- (ethS\x);

        \node (ethE\x) [block, right = of ethDots\x, xshift=-0.4cm] {$\MakeUppercase{eth}_{j+\ethBEnd}$};
        \draw [arrowParent] (ethE\x) -- (ethDots\x);

        % curly brace for state
        \draw [braceLR] (ethS\x.west |- ethS\x.south) --
            node (ethStateCoord\x) [anchor=north, text width=110pt, xshift=30pt, yshift=-10pt] {\footnotesize{
                \textsc{STATE} \\
                Headers: $BTC_{0 \cdots i+\x}$ \\
                Latest PoR: $ETH_{j+\ethLastEnd}$ \\
                %% $\Sigma$ Weight: $\Sigma^{i+\x}_{k=0} \text{W}(BTC_{k})$
                }}
            (ethS\x.east |- ethS\x.south);

        % curly brace for lots of eth blocks
        \draw [ethBraceR] (ethS\x.west |- ethS\x.north) -- node (ethAll\x) [black, midway, yshift=14pt] {
            %\scriptsize{$ETH_{40(j+\x) \cdots 40(j+\x)+39}$}
            \scriptsize{40 Eth Blocks}
        } (ethE\x.east |- ethE\x.north);

        % eth refl
        \draw [arrowRefl] (ethS\x) --
            node [anchor=center, above, sloped] {\scriptsize{include header}}
            node [anchor=center, below, sloped] {\scriptsize{and PoR}}
            (\oPrev);

        \fi
    }

    \node (botLeft) [below = of ethChain.west -| ethChain.north, xshift=-5pt, yshift=-10pt] {};
    \node (botRight) [below = of ethChain.west -| btc3.east, xshift=40pt, yshift=-10pt] {};
    \draw [arrowParent] (botLeft) node () [anchor=south, xshift=20pt] {\textit{Time}} -- (botRight);
