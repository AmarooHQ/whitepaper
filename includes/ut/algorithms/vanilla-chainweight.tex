\begin{algorithm}
\caption{Vanilla Chain Weight Algorithm}\label{alg:vanilla-bw}
\begin{algorithmic}
\Procedure{WeighChain}{$blocks$}\Comment{The weight of a chain}
  \If {length($blocks$) == 0}
    \State \Return{0}
  \EndIf
  \State \Return{\Call{WeighBlock}{head($blocks$)} + \Call{WeighChain}{tail($blocks$)}}
\EndProcedure
\\
\Procedure{WeighBlock}{$block$} \Comment{The weight of an individual block}
  \State \Return{\Call{LocalBlockWeight}{$block$}} \Comment{NB: defined by the consensus method}
\EndProcedure
\end{algorithmic}
\end{algorithm}
