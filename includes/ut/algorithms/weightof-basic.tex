\begin{algorithm}
\caption{A simplistic \textsc{WeightOf} for networks of a single root token}
\label{alg:weightof-1}
\begin{algorithmic}
\Procedure{WeightOf}{$B_i, state$} \Comment{The weight of a reflecting block in L-hashes}
    \State $t \gets$ \Call{BlockTimestamp}{$B_i$}
    \State $R \gets$ \Call{ChainOf}{$B_i$, $state$} \Comment{Either the local or a reflecting chain}
    \State $R_r \gets$ \Call{BlockRewardOfAt}{$R$, $t$, $state$} \Comment{L-coins/R-block --- by definition}
    \State $L_r \gets$ \Call{LocalBlockRewardAt}{$t$, $state$}
    \State $L_d \gets$ \Call{LocalBlockDifficultyAt}{$t$, $state$}
    \State \Return{$R_r \cdot (L_d \div L_r)$} \Comment{See \autoref{eq:por-conv-work-srt} (Note: $R_d$ cancels out)}
\EndProcedure
