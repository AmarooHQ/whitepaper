\begin{algorithm}
\caption{A simplistic \textsc{WeightOf} for reflections in a blockchain with a different root token}
\label{alg:weightof-dex}
\begin{algorithmic}
\Procedure{WeightOf}{$B_i, state$} \Comment{The weight of a reflecting block normalized to coins}
    \State $t \gets$ \Call{BlockTimestamp}{$B_i$}
    \State $C \gets$ \Call{ChainOf}{$B_i$, $state$}
    \State $r \gets$ \Call{BlockRewardOfChainAt}{$C$, $t$} \Comment{Block reward of $C$ at $t$}
    \State $\mathbb{C} \gets$ \Call{GetCoinDenomination}{$r$} \Comment{Native currency of $C$ (i.e., it's coin)}
    % \State $f \gets$ \Call{BlockFrequencyOfChain}{$C$} \Comment{Block production Hz of $C$}
    \State $x \gets$ \Call{ExchangeRateOfCoinToLocalAt}{$\mathbb{C}$, $t$}
    \State \Return{$x \cdot r$}
\EndProcedure
\end{algorithmic}
\end{algorithm}
