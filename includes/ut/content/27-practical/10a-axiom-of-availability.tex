
\DeclareAxiom{Availability}{%
    All reflected blocks must be verifiably complete and readily available.
}{%
    A block, \BL{}, cannot yet be valid unless, for each reflected block \BR{}:
    \begin{axiomreqs}
        \item \BR{} has been downloaded (i.e., the block is available); and
        \item the cryptographic integrity of \BR{} has been verified; and
        \item \BR{} is valid under this axiom.
    \end{axiomreqs}
}
