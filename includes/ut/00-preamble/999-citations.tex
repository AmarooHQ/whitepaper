%% %%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
%% %%%% Citations to command %%%%
%% %%%%%%%%%%%%%%%%%%%%%%%%%%%%%%

%% Make a new command for each citation that we want
%% e.g. \newcite{\citeGhostFull}{...content}
%% NOTE: for future reference -- had to separate the content line
%%      from the actual citation because expanding within the line
%%      was causing issues with compilation.
\newcommand{\newcite}[3]{
  \refstepcounter{citationnum}
  \addtocontents{citation}{\protect\contentsline{citation}{\thecitationnum. #2}{}{}}
  \expandafter\def\csname#1\endcsname{%
    #3
  }
}

%% %%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
%% %%%% Supplementary        %%%%
%% %%%%%%%%%%%%%%%%%%%%%%%%%%%%%%

%% Add a URL directly to the list (with a name)
%% will not return anything
%% \suppcitedirect{URL}{name}
\newcommand{\suppcitedirect}[2]{
  \refstepcounter{suppcitesnum}
  \addtocontents{suppcites}{\protect\contentsline{suppcites}{\thesuppcitesnum. {\href{#1}{#2}}}{}{}}
}

%% Reference a citation directly (will return a href)
%% \suppciteref{URL}{name}
\newcommand{\suppciteref}[2]{
  \refstepcounter{suppcitesnum}
  \addtocontents{suppcites}{\protect\contentsline{suppcites}{\thesuppcitesnum. {\href{#1}{#2}}}{}{}}
  \href{#1}{#2}
}

%% Create a new `\cite_____Link` command
%% \suppcitelink{commandid}{url}{table of contents name}
%% e.g. \suppcitelink{citeGhostLink}{https...}{Secure High-Rate Transaction Processing in Bitcoin}
\newcommand{\suppcitelink}[3]{
  \refstepcounter{suppcitesnum}
  \addtocontents{suppcites}{\protect\contentsline{suppcites}{\thesuppcitesnum. {\href{#2}{#3}}}{}{}}
  \expandafter\def\csname#1\endcsname{%
    #2
  }
}
