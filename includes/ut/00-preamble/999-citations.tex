%% Ignore all the page numbers and dots -- causing the latex
%% compilation to error.
\renewcommand{\cftcitationdotsep}{\cftnodots}
\renewcommand{\cftcitationpresnum}{\relax}
\renewcommand{\cftcitationaftersnum}{\hfill}

%% Make a new command for each citation that we want
%% e.g. \newcite{\citeGhostFull}{...content}
%% NOTE: for future reference -- had to separate the content line
%%      from the actual citation because expanding within the line
%%      was causing issues with compilation.
\newcommand{\newcite}[3]{
  \refstepcounter{citationnum}
  \addtocontents{citation}{\protect\contentsline{citation}{\thecitationnum. #2}{}{}}
  %% \addtocontents{citation}{\thecitationnum. #2 \\}
  \expandafter\def\csname#1\endcsname{%
    #3
  }
}
