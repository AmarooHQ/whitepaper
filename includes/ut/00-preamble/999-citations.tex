%% Key citation direct entry to the list
%% will not return, is used for manual citation additions.
%% Usage: \keyciteDirect{title}{url}{author}{date}
\newcommand{\keyciteDirect}[4]{
    % reference the counter to increment the number
    \refstepcounter{keycitationnum}
    % add the item to the contents line, we need to disable the page number / section number reference, so that it doesn't break.
    \addtocontents{%
        citation% what list are we adding to?
    }{%
        \protect\contentsline{citation}{%
            \thekeycitationnum. {{\href{#2}{#1}}, #3, #4.}%
        }{}{}
    }
}


%% Key citation create a new command
%% Will return a new command to use, which will provide the reference within
%% the text.
%% Usage: \keyciteCommand{commandName}{title}{url}{author}{date}
%% e.g. \keyciteCommand{helloworld}{MyTitle}{http://title.com}{Me}{2025}
%%          -> \helloworld
\newcommand{\keyciteCommand}[5]{
    \refstepcounter{keycitationnum}
    \addtocontents{%
        citation% what list are we adding to?
    }{%
        \protect\contentsline{citation}{%
            \thekeycitationnum. {{\href{#3}{#2}}, #4, #5.}%
        }{}{}
    }
    % Provide the command
    \expandafter\def\csname#1\endcsname{%
        {\href{#3}{#2}, #4, #5}%
    }
}

%% Supplementary citation direct entry to the list
%% will not return, is used for manual citation additions.
%% Usage: \suppciteDirect{title}{url}{author}{date}
\newcommand{\suppciteDirect}[4]{
    % reference the counter to increment the number
    \refstepcounter{suppcitesnum}
    % add the item to the contents line, we need to disable the page number / section number reference, so that it doesn't break.
    \addtocontents{%
        suppcites% what list are we adding to?
    }{%
        \protect\contentsline{suppcites}{%
            \thesuppcitesnum. {{\href{#2}{#1}}, #3, #4.}%
        }{}{}
    }
}

%% Supplementary citation - returns a new command that will provide ONLY
%% the URL that is passed in when called.
%% Usage: \suppciteUrlCommand{command}{title}{url}{author}{date}
%%      -> \command{}
%%              -> url
%% e.g. \suppciteUrlCommand{hello}{myweb}{http://my.web}{me}{2025}
%%      \hello -> http://my.web
\newcommand{\suppciteUrlCommand}[5]{
    \refstepcounter{suppcitesnum}
    \addtocontents{%
        suppcites% what list are we adding to?
    }{%
        \protect\contentsline{suppcites}{%
            \thesuppcitesnum. {{\href{#3}{#2}}, #4, #5.}%
        }{}{}
    }
    % Provide the command
    \expandafter\def\csname#1\endcsname{%
        #3% JUST the url (http://....)
    }
}

%% Supplementary citation create a new command
%% Will return a new command to use, which will provide the reference within
%% the text.
%% Usage: \keyciteCommand{commandName}{title}{url}{author}{date}
%% e.g. \keyciteCommand{helloworld}{MyTitle}{http://title.com}{Me}{2025}
%%          -> \helloworld -> \href{http://title.com}{MyTitle}
\newcommand{\suppciteHrefCommand}[5]{
    \refstepcounter{suppcitesnum}
    \addtocontents{%
        suppcites% what list are we adding to?
    }{%
        \protect\contentsline{suppcites}{%
          \thesuppcitesnum. {{\href{#3}{#2}}, #4, #5.}%
        }{}{}
    }
    % Provide the command
    \expandafter\def\csname#1\endcsname{%
        {\href{#3}{#2 (#4, #5)}}%
    }
}
