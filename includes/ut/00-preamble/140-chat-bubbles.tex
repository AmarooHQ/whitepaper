%% chat bubbles

\newcommand{\ChatRaw}[3]{
    \ifthenelse{\equal{}{#1}}{
        \begin{tcolorbox}[#2]
    }{
        \begin{tcolorbox}[#2,title=#1:]
    }
        \begin{varwidth}{0.8\textwidth}
            #3
        \end{varwidth}
    \end{tcolorbox}
}

\newcommand{\ChatL}[3]{
    \ChatRaw{#1}{hbox,#2}{#3}
}

\newcommand{\ChatR}[3]{
    \begin{flushright}
    \ChatRaw{#1}{hbox,#2}{#3}
    \end{flushright}
}

}
