\newcommand{\ChatRaw}[2]{

        #1
    ]
    \begin{varwidth}{0.8\textwidth}

    \end{varwidth}
    \end{tcolorbox}
}

    \ChatRaw{hbox,#1}{#2}
}

\newcommand{\ChatR}[2]{
    \begin{flushright}
    \ChatRaw{hbox,#1}{#2}
    \end{flushright}
}
