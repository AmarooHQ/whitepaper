%% preserve original quote cmd

\let\oldquote=\quote
\let\endoldquote=\endquote

%% quoting env using mdframed

%% -- a bit buggy use quote or bquote below

\newenvironment{mdquote}{\begin{mdframed}[
    innerleftmargin=1em,
    innertopmargin=.75em,
    innerbottommargin=.75em,
    skipabove=0.5em,
    backgroundcolor=gray!15,
    linewidth=1,
    leftline=true,
    rightline=false,
    topline=false,
    bottomline=false,
]}{\end{mdframed}}

%% less buggy quoting using tcolorbox

\RenewDocumentEnvironment{quote}{ o }{%
    \begin{tcolorbox}[
        colback=gray!15,colbacktitle=gray!15,
        tile,
        borderline west={0.75pt}{0pt}{black},
        % breakable,
        \IfNoValueTF{#1}{}{#1}
        ]
        \reversemarginpar\marginnote{\nullOnCopy{\textbf{Quote}}}
}{
    \end{tcolorbox}
}

%% conditionally add attribution to block quotes

\newcommand{\q}[1]{
    % directly copying quote env begin/end here -- not ideal, fixes build issues w/ hbox when it's provided as a parameter.
    % possibly related? https://github.com/T-F-S/tcolorbox/issues/154
    \begin{tcolorbox}[
        colback=gray!15,colbacktitle=gray!15,
        tile,
        borderline west={0.75pt}{0pt}{black},
        % breakable,
        hbox
        ]
    #1
    \end{tcolorbox}
}

\NewDocumentCommand{\bquote}{ m o }{
    \begin{quote}
    \reversemarginpar\marginnote{\nullOnCopy{\textbf{Quote}}}
    #1
    \IfNoValueTF{#2}{}{
        \par
        \noindent
        \xrfill[0.7ex]{0.5pt}
        \vspace{0em}
        \begin{flushright} \textit{--- #2} \end{flushright}
    }
    \end{quote}
}

\newenvironment{easide}{
    \begin{tcolorbox}[
        colback=gray!15,colbacktitle=gray!15,
        tile,
        breakable]%
    \reversemarginpar\marginnote{\nullOnCopy{\textbf{Aside}}}%
}{
    % \reversemarginpar\marginnote{\nullOnCopy{\textbf{Aside}}}%
    \end{tcolorbox}
}

\newcommand{\aside}[1]{
    \begin{easide}%
    #1
    \end{easide}
}

\newenvironment{easide*}{
    \begin{tcolorbox}[
        colback=gray!15,colbacktitle=gray!15,
        tile]
}{
    \end{tcolorbox}
}

\newcommand{\asideNB}[1]{
    \begin{easide*}
    #1
    \end{easide*}
}
